+#!/usr/bin/env python3
+"""PreToolUse skill staleness check.
+
+Fires on read_file calls. When the target path is a SKILL.md file,
+checks if the file's mtime is newer than the last-seen mtime stored
+in a per-session state file. If stale, emits a system reminder.
+
+Multi-terminal safe: per-session state files keyed by GROK_SESSION_ID.
+"""
+import json
+import os
+import sys
+from pathlib import Path
+
+def main():
+    try:
+        raw = sys.stdin.read()
+        data = json.loads(raw) if raw.strip() else {}
+    except Exception:
+        sys.exit(0)
+
+    tool_name = data.get("tool_name", "")
+    if tool_name != "read_file":
+        sys.exit(0)
+
+    # Check if target is a SKILL.md file
+    tool_input = data.get("tool_input", {})
+    target_file = tool_input.get("target_file", tool_input.get("file_path", ""))
+    if not target_file or "SKILL.md" not in str(target_file):
+        sys.exit(0)
+
+    target_path = Path(target_file)
+    if not target_path.exists():
+        sys.exit(0)
+
+    current_mtime = target_path.stat().st_mtime
+
+    # Per-session state file
+    session_id = os.environ.get("GROK_SESSION_ID", "unknown")
+    state_dir = Path.home() / ".grok" / "state" / session_id
+    state_dir.mkdir(parents=True, exist_ok=True)
+    state_file = state_dir / "skill-mtimes.json"
+
+    # Load stored mtimes
+    try:
+        stored = json.loads(state_file.read_text(encoding="utf-8"))
+    except Exception:
+        stored = {}
+
+    stored_key = str(target_path.resolve())
+    last_mtime = stored.get(stored_key, 0)
+
+    # Compare
+    if last_mtime > 0 and current_mtime > last_mtime:
+        # File changed since last read by this session
+        age = current_mtime - last_mtime
+        print(json.dumps({
+            "decision": "allow",
+            "hookSpecificOutput": {
+                "hookEvent": "PreToolUse",
+                "additionalContext": (
+                    f"⚠️ Skill file changed since your last read "
+                    f"({age:.0f}s ago): {target_path.name}. "
+                    f"Another session may have edited it. "
+                    f"The version you're about to read is newer than "
+                    f"what you may have cached. Read it carefully."
+                )
+            }
+        }))
+    elif last_mtime == 0:
+        # First read — just store
+        pass
+
+    # Update stored mtime
+    stored[stored_key] = current_mtime
+    try:
+        tmp = state_file.with_suffix(f".{os.getpid()}.tmp")
+        tmp.write_text(json.dumps(stored, indent=2), encoding="utf-8")
+        tmp.replace(state_file)
+    except Exception:
+        pass
+
+    sys.exit(0)
+
+if __name__ == "__main__":
+    main()

+"""claim_handoff.py — Claim a handoff for fleet coordination.
+
+Sets the assigned_to/assigned_at/assigned_by fields in the YAML frontmatter
+and appends a changelog row. If already claimed by a different active session,
+warns and exits non-zero.
+
+Usage:
+    python claim_handoff.py <path> [--session <id>] [--host <host>] [--release]
+
+    --session   Session ID (defaults to $GROK_SESSION_ID or 'unknown')
+    --host      Host identifier (defaults to 'grok')
+    --release   Release the claim instead of claiming
+"""
+
+from __future__ import annotations
+
+import argparse
+import os
+import re

+

+def _utc_now() -> str:
+    return datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M")
+
+
+def _parse_frontmatter(text: str) -> tuple[str, str, str]:
+    """Split into pre-fm, fm, post-fm. Returns (before, fm_block, after)."""
+    if not text.startswith("---"):
+        return "", "", text
+    end = text.find("\n---", 3)
+    if end == -1:
+        return "", "", text
+    fm = text[:end + 4]
+    after = text[end + 4:]
+    return "", fm, after
+
+
+def _get_field(fm: str, field: str) -> str:
+    """Extract a YAML field value from frontmatter text."""
+    m = re.search(rf"^#?\s*{field}:\s*(.+)$", fm, re.MULTILINE)
+    if m:
+        val = m.group(1).strip()
+        # Strip inline comments
+        if "#" in val:
+            val = val.split("#")[0].strip()
+        # Strip quotes
+        val = val.strip('"').strip("'")
+        return val
+    return ""
+
+
+def _set_field(fm: str, field: str, value: str) -> str:
+    """Set or replace a YAML field in frontmatter text."""
+    # Try to replace existing (commented or uncommented) line
+    pattern = rf"^(#?\s*){re.escape(field)}:\s*.+$"
+    replacement = f"{field}: {value}"
+    if re.search(pattern, fm, re.MULTILINE):
+        return re.sub(pattern, replacement, fm, flags=re.MULTILINE)
+    # Field doesn't exist — add before closing ---
+    return fm.rstrip() + f"\n{field}: {value}\n---\n"
+
+
+def _remove_field(fm: str, field: str) -> str:
+    """Remove a YAML field (commented or uncommented) from frontmatter."""
+    pattern = rf"^#?\s*{re.escape(field)}:\s*.+\n?"
+    return re.sub(pattern, "", fm, flags=re.MULTILINE)
+
+
+def _append_changelog(text: str, session: str, action: str) -> str:
+    """Append a row to the changelog table, or create the table if missing."""
+    now = _utc_now()
+    row = f"| {now} | {session[:8]}... | {action} |"
+
+    # Check if changelog table exists
+    cl_match = re.search(r"## Changelog\s*\n\s*\|.*\n\|[-:| ]+\n", text, re.IGNORECASE)
+    if cl_match:
+        # Append row after the header separator line
+        end_of_header = cl_match.end()
+        return text[:end_of_header] + row + "\n" + text[end_of_header:]
+    else:
+        # Create changelog at end of file
+        return text.rstrip() + f"\n\n## Changelog\n\n| Date | Session | Action |\n|------|---------|--------|\n{row}\n"
+
+
+def claim(path: Path, session: str, host: str) -> int:
+    """Claim a handoff file."""
+    if not path.exists():
+        print(f"ERROR: handoff not found: {path}", file=sys.stderr)
+        return 1
+
+    text = path.read_text(encoding="utf-8")
+
+    # Check if already claimed
+    existing = _get_field(text, "assigned_to")
+    existing_by = _get_field(text, "assigned_by")
+    if existing and existing_by and existing_by != session:
+        existing_short = existing_by[:8] if len(existing_by) >= 8 else existing_by
+        print(f"WARNING: already claimed by '{existing}' (session {existing_short}...)", file=sys.stderr)
+        print("If that session is stale, use --release first.", file=sys.stderr)
+        return 2
+
+    now = _utc_now()
+
+    # Set assignment fields
+    text = _set_field(text, "assigned_to", host)
+    text = _set_field(text, "assigned_at", now)
+    text = _set_field(text, "assigned_by", session)
+
+    # Append changelog
+    text = _append_changelog(text, session, f"claimed by {host}")
+
+    path.write_text(text, encoding="utf-8")
+    print(f"Claimed: {path.name} → assigned_to={host}, assigned_by={session[:8]}...")
+    return 0
+
+
+def release(path: Path, session: str) -> int:
+    """Release a claim on a handoff file."""
+    if not path.exists():
+        print(f"ERROR: handoff not found: {path}", file=sys.stderr)
+        return 1
+
+    text = path.read_text(encoding="utf-8")
+
+    existing = _get_field(text, "assigned_to")
+    if not existing:
+        print("Not claimed — nothing to release.", file=sys.stderr)
+        return 0
+
+    # Remove assignment fields
+    text = _remove_field(text, "assigned_to")
+    text = _remove_field(text, "assigned_at")
+    text = _remove_field(text, "assigned_by")
+
+    # Append changelog
+    text = _append_changelog(text, session, "claim released")
+
+    path.write_text(text, encoding="utf-8")
+    print(f"Released claim on: {path.name}")
+    return 0
+
+
+def main() -> int:
+    parser = argparse.ArgumentParser(description="Claim or release a handoff")
+    parser.add_argument("path", help="Path to HANDOFF.md")
+    parser.add_argument("--session", default=os.environ.get("GROK_SESSION_ID", "unknown"),
+                        help="Session ID claiming the handoff")
+    parser.add_argument("--host", default="grok",
+                        help="Host identifier (default: grok)")
+    parser.add_argument("--release", action="store_true",
+                        help="Release the claim instead of claiming")
+    args = parser.parse_args()
+
+    path = Path(args.path).resolve()
+    if args.release:
+        return release(path, args.session)
+    else:
+        return claim(path, args.session, args.host)
+
+
+if __name__ == "__main__":
+    sys.exit(main())
