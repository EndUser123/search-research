    id INTEGER PRIMARY KEY AUTOINCREMENT,

    change_type TEXT NOT NULL,       -- bug_fix, feature, refactor, breaking, docs

);

-- Artifact status per pending item
CREATE TABLE artifact_status (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    pending_item_id INTEGER NOT NULL,
    artifact_type TEXT NOT NULL,     -- changelog, prd, ard, readme
    status TEXT NOT NULL,            -- pending, confirm, done, n/a
    updated_at TEXT DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (pending_item_id) REFERENCES pending_items(id),
    UNIQUE(pending_item_id, artifact_type)
);

"""
artifact_severity.py - Impact-based severity scoring
"""

from pathlib import Path
import fnmatch

# File patterns → severity mapping
SEVERITY_PATTERNS = {
    "critical": [
        "src/api/**",
        "src/schema/**", 
        "src/core/**",
        "src/engine/**",
        "**/models.py",
        "**/schemas.py",
        "pyproject.toml",
        "setup.py",
    ],
    "standard": [
        "src/features/**",
        "src/commands/**",
        "src/hooks/**",
        "src/modules/**",
        "src/**/*.py",  # Catch-all for src
    ],
    "low": [
        "tests/**",
        "docs/**",
        "scripts/**",
        "*.md",
        ".github/**",
        "examples/**",
    ],
}

# Severity → artifact requirements
SEVERITY_REQUIREMENTS = {
    "critical": {
        "changelog": "required",
        "prd": "required",
        "ard": "required",
        "readme": "confirm",
    },
    "standard": {
        "changelog": "required",
        "prd": "confirm",
        "ard": "confirm",
        "readme": "skip",
    },
    "low": {
        "changelog": "confirm",
        "prd": "skip",
        "ard": "skip",
        "readme": "skip",
    },
}

# Change type modifiers (applied on top of severity)
CHANGE_TYPE_MODIFIERS = {
    "breaking": {
        # Breaking changes escalate everything to required
        "changelog": "required",
        "prd": "required",
        "ard": "required",
        "readme": "required",
    },
    "bug_fix": {
        # Bug fixes usually don't need PRD/ARD
        "prd": "skip",
        "ard": "skip",
    },
    "docs": {
        # Doc changes rarely need artifacts
        "changelog": "skip",
        "prd": "skip",
        "ard": "skip",
        "readme": "confirm",
    },
}


def calculate_severity(files_changed: list[str]) -> str:
    """
    Calculate severity based on highest-impact file changed.
    Returns: 'critical' | 'standard' | 'low'
    """
    for severity in ["critical", "standard", "low"]:
        patterns = SEVERITY_PATTERNS[severity]
        for file_path in files_changed:
            for pattern in patterns:
                if fnmatch.fnmatch(file_path, pattern):
                    if severity == "critical":
                        return "critical"  # Short-circuit on first critical
                    elif severity == "standard":
                        # Continue checking for critical, but mark standard found
                        pass
    
    # More precise: check each file, return highest severity found
    highest = "low"
    severity_rank = {"critical": 3, "standard": 2, "low": 1}
    
    for file_path in files_changed:
        for severity, patterns in SEVERITY_PATTERNS.items():
            for pattern in patterns:
                if fnmatch.fnmatch(file_path, pattern):
                    if severity_rank[severity] > severity_rank[highest]:
                        highest = severity
                    break
    
    return highest


def get_artifact_requirements(
    change_type: str,
    severity: str,
) -> dict[str, str]:
    """
    Get artifact requirements based on severity and change type.
    Returns: {artifact_type: 'required' | 'confirm' | 'skip'}
    """
    # Start with severity-based requirements
    requirements = SEVERITY_REQUIREMENTS.get(severity, SEVERITY_REQUIREMENTS["standard"]).copy()
    
    # Apply change type modifiers
    modifiers = CHANGE_TYPE_MODIFIERS.get(change_type, {})
    for artifact_type, requirement in modifiers.items():
        # Modifiers can only escalate or skip, not downgrade required
        current = requirements.get(artifact_type, "skip")
        if requirement == "required":
            requirements[artifact_type] = "required"
        elif requirement == "skip" and current != "required":
            requirements[artifact_type] = "skip"
        elif requirement == "confirm" and current == "skip":
            requirements[artifact_type] = "confirm"
    
    return requirements


def infer_artifact_needs(
    change_type: str,
    files_changed: list[str],
) -> tuple[str, dict[str, str]]:
    """
    Full inference: calculate severity, then requirements.
    Returns: (severity, {artifact_type: 'pending' | 'confirm' | 'n/a'})
    """
    severity = calculate_severity(files_changed)
    requirements = get_artifact_requirements(change_type, severity)
    
    # Map requirements to tracking status
    status_map = {
        "required": "pending",
        "confirm": "confirm",
        "skip": "n/a",
    }
    
    return severity, {k: status_map[v] for k, v in requirements.items()}

"""
artifact_git_verify.py - Pre-commit verification
"""

import sys
import sqlite3
from pathlib import Path

DB_PATH = Path("P:/.claude/artifacts.db")

def get_blocking_items(project_root: str) -> list[dict]:
    """Get items that should block commit."""
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    
    # Find pending items with required artifacts not done
    query = """
        SELECT p.item_id, p.summary, p.change_type, p.severity, a.artifact_type
        FROM pending_items p
        JOIN artifact_status a ON p.id = a.pending_item_id
        WHERE p.project_root = ?
          AND a.status = 'pending'
          AND (
              -- Critical severity: all artifacts required
              (p.severity = 'critical')
              -- Breaking changes: all artifacts required  
              OR (p.change_type = 'breaking')
              -- Standard/low: only changelog required for bug_fix/feature
              OR (p.change_type IN ('bug_fix', 'feature') AND a.artifact_type = 'changelog')
          )
    """
    
    cursor = conn.execute(query, (project_root,))
    results = [dict(row) for row in cursor.fetchall()]
    conn.close()
    
    return results


def main():
    import argparse
    parser = argparse.ArgumentParser()
    parser.add_argument("--staged", action="store_true")
    args = parser.parse_args()
    
    # Determine project root
    project_root = Path.cwd().resolve()
    while project_root != project_root.parent:
        if (project_root / ".git").exists():
            break
        project_root = project_root.parent

"""
artifact_session.py - Multi-instance session management
"""

import sqlite3
import uuid
import os
from datetime import datetime, timezone, timedelta
from pathlib import Path
from contextlib import contextmanager

DB_PATH = Path("P:/.claude/artifacts.db")
SESSION_TIMEOUT_MINUTES = 30


def get_session_id() -> str:
    """Get or create session ID for this instance."""
    # Use environment variable if set (allows explicit session naming)
    if "CLAUDE_SESSION_ID" in os.environ:
        return os.environ["CLAUDE_SESSION_ID"]
    
    # Generate persistent ID for this process
    if not hasattr(get_session_id, "_id"):
        get_session_id._id = f"session-{uuid.uuid4().hex[:8]}"
    
    return get_session_id._id


def register_session(project_root: str, instance_name: str = None) -> str:
    """Register this session in the database."""
    session_id = get_session_id()
    instance_name = instance_name or f"claude-{os.getpid()}"
    
    conn = sqlite3.connect(DB_PATH)
    conn.execute("""
        INSERT OR REPLACE INTO sessions (session_id, instance_name, project_root, last_heartbeat)
        VALUES (?, ?, ?, ?)
    """, (session_id, instance_name, project_root, datetime.now(timezone.utc).isoformat()))
    conn.commit()
    conn.close()
    
    return session_id


def heartbeat():
    """Update session heartbeat."""
    session_id = get_session_id()
    conn = sqlite3.connect(DB_PATH)
    conn.execute("""
        UPDATE sessions SET last_heartbeat = ? WHERE session_id = ?
    """, (datetime.now(timezone.utc).isoformat(), session_id))
    conn.commit()
    conn.close()


def cleanup_stale_sessions():
    """Remove sessions that haven't heartbeated recently."""
    cutoff = (datetime.now(timezone.utc) - timedelta(minutes=SESSION_TIMEOUT_MINUTES)).isoformat()
    
    conn = sqlite3.connect(DB_PATH)
    # Release locks held by stale sessions
    conn.execute("""
        UPDATE pending_items 
        SET locked_by_session = NULL
        WHERE locked_by_session IN (
            SELECT session_id FROM sessions WHERE last_heartbeat < ?
        )
    """, (cutoff,))
    # Remove stale sessions
    conn.execute("DELETE FROM sessions WHERE last_heartbeat < ?", (cutoff,))
    conn.commit()
    conn.close()


@contextmanager
def item_lock(item_id: str, project_root: str):
    """
    Context manager to lock a pending item for exclusive access.
    Prevents race conditions when multiple instances modify same item.
    """
    session_id = get_session_id()
    conn = sqlite3.connect(DB_PATH, timeout=10)
    
    try:
        # Try to acquire lock
        cursor = conn.execute("""
            UPDATE pending_items 
            SET locked_by_session = ?
            WHERE item_id = ? AND project_root = ?
              AND (locked_by_session IS NULL OR locked_by_session = ?)
        """, (session_id, item_id, project_root, session_id))
        
        if cursor.rowcount == 0:
            # Check if item exists but is locked by another session
            cursor = conn.execute("""
                SELECT locked_by_session FROM pending_items
                WHERE item_id = ? AND project_root = ?
            """, (item_id, project_root))
            row = cursor.fetchone()
            if row and row[0] and row[0] != session_id:
                raise RuntimeError(f"Item {item_id} locked by another session: {row[0]}")
        
        conn.commit()
        yield conn
        
    finally:
        # Release lock
        conn.execute("""
            UPDATE pending_items 
            SET locked_by_session = NULL
            WHERE item_id = ? AND project_root = ? AND locked_by_session = ?
        """, (item_id, project_root, session_id))
        conn.commit()
        conn.close()

def format_ago(delta: timedelta) -> str:
    """Format timedelta as human-readable string."""
    seconds = int(delta.total_seconds())
    if seconds < 3600:
        return f"{seconds // 60} minutes ago"
    elif seconds < 86400:
        return f"{seconds // 3600} hours ago"
    elif seconds < 604800:
        return f"{seconds // 86400} days ago"
    else:
        return f"{seconds // 604800} weeks ago"
