```python
from __future__ import annotations

from pathlib import Path
from typing import Any

from .evidence_packet import CoverageReport, EvidencePacket, SearchMatch
from .search_plan import SearchPlan


def _expand_root(root: str) -> str:
    return str(Path(root.replace("~", str(Path.home()))).expanduser())


def execute_plan(plan: SearchPlan, search_fn) -> EvidencePacket:
    searched_roots: list[str] = []
    matches: list[SearchMatch] = []
    backend_failures: list[str] = []

    for root in plan.required_roots:
        expanded = _expand_root(root)
        try:
            results: list[dict[str, Any]] = search_fn(
                query=plan.entity_id or plan.query,
                root=expanded,
                backends=list(plan.required_backends),
            )
            searched_roots.append(root)
            for r in results:
                matches.append(
                    SearchMatch(
                        source_path=r.get("filepath", ""),
                        backend=r.get("source", "unknown"),
                        title=r.get("title", ""),
                        snippet=r.get("content", "")[:300],
                    )
                )
        except Exception as exc:
            backend_failures.append(f"{root}: {exc!s}")

    missing_roots = tuple(r for r in plan.required_roots if r not in searched_roots)
    coverage = CoverageReport(
        required_roots=plan.required_roots,
        searched_roots=tuple(searched_roots),
        missing_roots=missing_roots,
        complete=len(missing_roots) == 0,
    )

    return EvidencePacket(
        query=plan.query,
        intent_kind=plan.intent_kind,
        entity_type=plan.entity_type,
        entity_id=plan.entity_id,
        coverage=coverage,
        matches=tuple(matches),
        backend_failures=tuple(backend_failures),
    )
```

```python
import re
from dataclasses import dataclass
from typing import Literal

from .entity_types import EntityType

HASH_SKILL_RE = re.compile(r"\b[\w.-]+-skill-[0-9a-f]{8}\b", re.IGNORECASE)
SEARCH_INTENT_RE = re.compile(
    r"\b(where is|find|search for|look for|does .* exist|can you find|locate)\b",
    re.IGNORECASE,
)


@dataclass(frozen=True)
class SearchIntent:
    kind: Literal["entity_existence", "entity_location", "content_search", "generic"]
    confidence: float
    entity_type: EntityType = EntityType.UNKNOWN
    entity_id: str | None = None


def classify_intent(user_text: str) -> SearchIntent:
    text = user_text.strip()

    match = HASH_SKILL_RE.search(text)
    if match:
        return SearchIntent(
            kind="entity_location",
            confidence=0.99,
            entity_type=EntityType.HASH_SKILL_ARTIFACT,
            entity_id=match.group(0),
        )

    if SEARCH_INTENT_RE.search(text):
        lowered = text.lower()
        if "command" in lowered:
            return SearchIntent("entity_location", 0.9, EntityType.COMMAND)
        if "skill" in lowered:
            return SearchIntent("entity_location", 0.9, EntityType.SKILL)
        if "plugin" in lowered:
            return SearchIntent("entity_location", 0.9, EntityType.PLUGIN)
        if "hook" in lowered:
            return SearchIntent("entity_location", 0.9, EntityType.HOOK)
        return SearchIntent("content_search", 0.65)

    return SearchIntent("generic", 0.4)
```

This must be the only module allowed to call `search-research` APIs. The search-research package has a broad and historically unstable surface area, so isolating it behind one adapter is mandatory.[1]

```python
from __future__ import annotations

from pathlib import Path
from typing import Any, Callable

from .evidence_packet import CoverageReport, EvidencePacket, SearchMatch
from .search_plan import SearchPlan

SearchFn = Callable[..., list[dict[str, Any]]]


def _expand_root(root: str) -> str:
    return str(Path(root.replace("~", str(Path.home()))).expanduser())


def execute_plan(plan: SearchPlan, search_fn: SearchFn) -> EvidencePacket:
    searched_roots: list[str] = []
    matches: list[SearchMatch] = []
    backend_failures: list[str] = []

    for root in plan.required_roots:
        expanded = _expand_root(root)
        try:
            results: list[dict[str, Any]] = search_fn(
                query=plan.entity_id or plan.query,
                root=expanded,
                backends=list(plan.required_backends),
            )
            searched_roots.append(root)
            for r in results:
                matches.append(
                    SearchMatch(
                        source_path=r.get("filepath", ""),
                        backend=r.get("source", "unknown"),
                        title=r.get("title", ""),
                        snippet=r.get("content", "")[:300],
                    )
                )
        except Exception as exc:
            backend_failures.append(f"{root}: {exc!s}")

    missing_roots = tuple(r for r in plan.required_roots if r not in searched_roots)
    coverage = CoverageReport(
        required_roots=plan.required_roots,
        searched_roots=tuple(searched_roots),
        missing_roots=missing_roots,
        complete=len(missing_roots) == 0,
    )

    return EvidencePacket(
        query=plan.query,
        intent_kind=plan.intent_kind,
        entity_type=plan.entity_type,
        entity_id=plan.entity_id,
        coverage=coverage,
        matches=tuple(matches),
        backend_failures=tuple(backend_failures),
    )
```

```python
        lines.append("matches:")
        for match in packet.matches:
            lines.append(f"- {match.source_path} [{match.backend}]")
    else:
        lines.append("matches: none")

    if packet.backend_failures:
        lines.append("backend_failures:")
        for failure in packet.backend_failures:
            lines.append(f"- {failure}")

    return "\n".join(lines)
```

### lib/intent_classifier.py

The classifier must be deterministic and narrow because the evidence shows prior intent detection was insufficient and misrouting risk was explicitly identified.[1]

```python
from __future__ import annotations

import re
from dataclasses import dataclass
from typing import Literal

from .entity_types import EntityType

HASH_SKILL_RE = re.compile(r"\b[\w.-]+-skill-[0-9a-f]{8}\b", re.IGNORECASE)
SEARCH_INTENT_RE = re.compile(
    r"\b(where is|find|search for|look for|does .* exist|can you find|locate)\b",
    re.IGNORECASE,
)


@dataclass(frozen=True)
class SearchIntent:
    kind: Literal["entity_existence", "entity_location", "content_search", "generic"]
    confidence: float
    entity_type: EntityType = EntityType.UNKNOWN
    entity_id: str | None = None


def classify_intent(user_text: str) -> SearchIntent:
    text = user_text.strip()

    match = HASH_SKILL_RE.search(text)
    if match:
        return SearchIntent(
            kind="entity_location",
            confidence=0.99,
            entity_type=EntityType.HASH_SKILL_ARTIFACT,
            entity_id=match.group(0),
        )

    if SEARCH_INTENT_RE.search(text):
        lowered = text.lower()
        if "command" in lowered:
            return SearchIntent("entity_location", 0.9, EntityType.COMMAND)
        if "skill" in lowered:
            return SearchIntent("entity_location", 0.9, EntityType.SKILL)
        if "plugin" in lowered:
            return SearchIntent("entity_location", 0.9, EntityType.PLUGIN)
        if "hook" in lowered:
            return SearchIntent("entity_location", 0.9, EntityType.HOOK)
        return SearchIntent("content_search", 0.65)

    return SearchIntent("generic", 0.4)
```

### lib/search_plan.py

```python
from __future__ import annotations

from dataclasses import dataclass

from .entity_scope_map import ENTITY_SCOPE_MAP
from .intent_classifier import SearchIntent


@dataclass(frozen=True)
class SearchPlan:
    query: str
    intent_kind: str
    entity_type: str
    entity_id: str | None
    required_roots: tuple[str, ...]
    optional_roots: tuple[str, ...]
    required_backends: tuple[str, ...]
    allow_partial_results: bool
    absence_claim_requires_full_coverage: bool


def build_search_plan(query: str, intent: SearchIntent) -> SearchPlan:
    required_roots = ENTITY_SCOPE_MAP[intent.entity_type]

    if intent.kind in {"entity_existence", "entity_location"}:
        return SearchPlan(
            query=query,
            intent_kind=intent.kind,
            entity_type=intent.entity_type.value,
            entity_id=intent.entity_id,
            required_roots=required_roots,
            optional_roots=(),
            required_backends=("grep", "skills"),
            allow_partial_results=False,
            absence_claim_requires_full_coverage=True,
        )

    return SearchPlan(
        query=query,
        intent_kind=intent.kind,
        entity_type=intent.entity_type.value,
        entity_id=intent.entity_id,
        required_roots=(),
        optional_roots=(),
        required_backends=("grep",),
        allow_partial_results=True,
        absence_claim_requires_full_coverage=False,
    )
```

```python
from dataclasses import dataclass, field


@dataclass(frozen=True)
class SearchMatch:
    source_path: str
    backend: str
    title: str
    snippet: str


@dataclass(frozen=True)
class CoverageReport:
    required_roots: tuple[str, ...]
    searched_roots: tuple[str, ...]
    missing_roots: tuple[str, ...]
    complete: bool


@dataclass(frozen=True)
class EvidencePacket:
    query: str
    intent_kind: str
    entity_type: str
    entity_id: str | None
    coverage: CoverageReport
    matches: tuple[SearchMatch, ...] = field(default_factory=tuple)
    backend_failures: tuple[str, ...] = field(default_factory=tuple)

    @property
    def absence_claim_allowed(self) -> bool:
        return self.coverage.complete and not self.backend_failures
```

### lib/searchresearch_bridge.py

This must be the only module allowed to call `search-research` APIs. The search-research package has a broad and historically unstable surface area, so isolating it behind one adapter is mandatory.[1]

```python
from __future__ import annotations

from pathlib import Path
from typing import Any, Callable

from .evidence_packet import CoverageReport, EvidencePacket, SearchMatch
from .search_plan import SearchPlan

SearchFn = Callable[..., list[dict[str, Any]]]


def _expand_root(root: str) -> str:
    return str(Path(root.replace("~", str(Path.home()))).expanduser())


def execute_plan(plan: SearchPlan, search_fn: SearchFn) -> EvidencePacket:
    searched_roots: list[str] = []
    matches: list[SearchMatch] = []
    backend_failures: list[str] = []

    for root in plan.required_roots:
        expanded = _expand_root(root)
        try:
            results: list[dict[str, Any]] = search_fn(
                query=plan.entity_id or plan.query,
                root=expanded,
                backends=list(plan.required_backends),
            )
            searched_roots.append(root)
            for r in results:
                matches.append(
                    SearchMatch(
                        source_path=r.get("filepath", ""),
                        backend=r.get("source", "unknown"),
                        title=r.get("title", ""),
                        snippet=r.get("content", "")[:300],
                    )
                )
        except Exception as exc:
            backend_failures.append(f"{root}: {exc!s}")

    missing_roots = tuple(r for r in plan.required_roots if r not in searched_roots)
    coverage = CoverageReport(
        required_roots=plan.required_roots,
        searched_roots=tuple(searched_roots),
        missing_roots=missing_roots,
        complete=len(missing_roots) == 0,
    )

    return EvidencePacket(
        query=plan.query,
        intent_kind=plan.intent_kind,
        entity_type=plan.entity_type,
        entity_id=plan.entity_id,
        coverage=coverage,
        matches=tuple(matches),
        backend_failures=tuple(backend_failures),
    )
```

### lib/packet_rendering.py

```python
from __future__ import annotations

from .evidence_packet import EvidencePacket


def render_packet_markdown(packet: EvidencePacket) -> str:
    lines: list[str] = []
    lines.append("[SEARCH EVIDENCE]")
    lines.append(f"query: {packet.query}")
    lines.append(f"intent_kind: {packet.intent_kind}")
    lines.append(f"entity_type: {packet.entity_type}")
    lines.append(f"entity_id: {packet.entity_id or 'n/a'}")
    lines.append(f"coverage_complete: {packet.coverage.complete}")
    lines.append(f"required_roots: {', '.join(packet.coverage.required_roots) or 'n/a'}")
    lines.append(f"searched_roots: {', '.join(packet.coverage.searched_roots) or 'n/a'}")
    lines.append(f"missing_roots: {', '.join(packet.coverage.missing_roots) or 'n/a'}")
    lines.append(f"absence_claim_allowed: {packet.absence_claim_allowed}")

    if packet.matches:
        lines.append("matches:")
        for match in packet.matches:
            lines.append(f"- {match.source_path} [{match.backend}]")
    else:
        lines.append("matches: none")

    if packet.backend_failures:
        lines.append("backend_failures:")
        for failure in packet.backend_failures:
            lines.append(f"- {failure}")

    return "\n".join(lines)
```

### hooks/Stop_claim_verifier.py

This hook is mandatory. The system failed previously because unsupported universal absence claims were allowed through, and the package must prevent that class of error directly.[1]

```python
from __future__ import annotations

import re

UNIVERSAL_ABSENCE_RE = re.compile(
    r"\b(not (?:on disk|found) anywhere|doesn't exist anywhere|not anywhere on disk)\b",
    re.IGNORECASE,
)


def run(input_data: dict) -> dict:
    response = str(input_data.get("response") or "")
    evidence = input_data.get("search_evidence_packet") or {}

    absence_claim_allowed = bool(evidence.get("absence_claim_allowed"))

    if UNIVERSAL_ABSENCE_RE.search(response) and not absence_claim_allowed:
        return {
            "decision": "block",
            "reason": (
                "Universal absence claim is not allowed because required search coverage "
                "was incomplete or unverified."
            ),
        }

    return {"decision": "allow"}
```

## Required integration behavior

The implementing LLM must preserve these rules:

1. Claude-specific entity semantics live in this plugin package, not in `search-research`.[1]
2. All `search-research` calls go through `lib/searchresearch_bridge.py`.[1]
3. All entity existence or location queries must produce an `EvidencePacket` before answer generation.[1]
4. Any universal absence wording must be blocked unless `absence_claim_allowed` is true.[1]
5. Partial results are acceptable for exploratory search but never sufficient for universal absence claims.[1]

## Implementation sequence

Use this order, because the attached evidence shows past problems with integration fragility, insufficient test realism, and overly broad refactors without clear safety rails.[1]

1. Create repo layout and static files.
2. Implement `entity_types`, `entity_scope_map`, `intent_classifier`, and `search_plan`.
3. Implement `evidence_packet` and `claude_entity_resolver`.
4. Implement `searchresearch_bridge` with a fake adapter first.
5. Write and pass all tests using fake search functions.
6. Wire `UserPromptSubmit_entity_scope` and `Stop_claim_verifier`.
7. Add the real `search-research` call path inside the bridge only after tests are green.
8. Add observability via `PostToolUse_evidence_logger`.
9. Add one end-to-end test for the original failure class: repo root search misses file in `~/.claude/commands/`, and the package forbids universal absence.

## Acceptance criteria

The package is complete only when all of the following are true:

- A hash-suffixed Claude Code artifact query resolves to `~/.claude/commands/` automatically.[1]
- Entity queries generate a `SearchPlan` with required roots and full-coverage requirements.[1]
- The bridge returns an `EvidencePacket` with explicit coverage reporting.[1]
- Universal absence claims are blocked when coverage is incomplete.[1]
- Tests cover the exact historical failure mode and pass reliably.[1]
- Claude-specific rules do not leak into generic `search-research` modules except through the bridge boundary.[1]

## Notes for the implementing LLM

Do not simplify this into fuzzy prompt engineering. The point of this package is to move truthfulness from prose heuristics into typed runtime contracts.[1]

Do not add broad “smart” context injection logic. Render the evidence packet and constrain the answer with it. The prior system already had too much interpretive behavior and insufficiently explainable failure modes.[1]

Do not spread `search-research` imports across the package. Keep them isolated behind `lib/searchresearch_bridge.py` so future changes in that codebase do not cascade through the plugin.[1]

## Phase Two

Phase One establishes deterministic entity resolution, mandatory scope expansion, evidence packets, coverage reporting, and claim verification. That solves the high-risk truthfulness problem for Claude Code entities, but it does not yet provide an optimal multi-store prompt context pipeline. The attached research strongly suggests that richer hybrid context capabilities exist in the broader system, including session memory, RAG integration, intelligent filtering, semantic bridging, and integration-quality scoring, but they are not yet defined as a single production context assembly contract.[1]

Phase Two adds that missing layer.

### Phase Two goals

Build a context assembly layer that takes results from Claude-local stores and search-research backends, ranks and compresses them, resolves conflicts, fits them into a prompt budget, and produces a prompt-ready structured context object for the LLM.[1]

### Design principles

1. **Prompt context is a product, not a dump**. Retrieval output must be transformed before it reaches the LLM.[1]
2. **Ranking objective differs from search objective**. Prompt context should optimize for answer utility per token, not just backend relevance score.[1]
3. **Conflicts must be surfaced, not averaged away**. The user explicitly values uncertainty disclosure, and the broader system materials repeatedly warn about hidden failure modes and graceful degradation masking real gaps.[1]
4. **Budget is explicit**. Each source class gets token budget constraints so one noisy backend cannot crowd out better evidence.[1]
5. **Traceability is mandatory**. Every injected fact must retain source provenance to support later answer validation.[1]

### Source classes

Phase Two should normalize evidence into these source classes:

- `entity_scope`: mandatory Claude Code filesystem and plugin-scope results
- `project_local`: repo-local files and indexes
- `session_memory`: recent session patterns and reconstructed context
- `knowledge_store`: CKS, KG, RAG, graph, embeddings-backed stores
- `search_backend`: search-research local backend outputs
- `web_external`: external provider outputs when allowed
- `policy`: package rules, plugin rules, search contract, and absence constraints

### context_ranker.py

```python
from __future__ import annotations

from .source_authority import SOURCE_AUTHORITY
from .synthesis_context import ContextFact


def score_fact(fact: ContextFact) -> float:
    authority_weight = SOURCE_AUTHORITY.get(fact.source_class, 0.5)
    return (
        authority_weight * 0.35
        + fact.relevance * 0.35
        + fact.novelty * 0.15
        + fact.recency * 0.15
    )


def rank_facts(facts: list[ContextFact]) -> list[ContextFact]:
    return sorted(facts, key=score_fact, reverse=True)
```

```python
from __future__ import annotations

from collections import defaultdict

from .context_ranker import score_fact
from .synthesis_context import ContextConflict, ContextFact


def resolve_conflicts(facts: list[ContextFact]) -> tuple[list[ContextFact], list[ContextConflict]]:
    grouped: dict[str, list[ContextFact]] = defaultdict(list)
    survivors: list[ContextFact] = []
    conflicts: list[ContextConflict] = []

    for fact in facts:
        if fact.conflict_group:
            grouped[fact.conflict_group].append(fact)
        else:
            survivors.append(fact)

    for group_id, members in grouped.items():
        ranked = sorted(members, key=score_fact, reverse=True)
        top = ranked[0]
        survivors.append(top)
        conflicts.append(
            ContextConflict(
                group_id=group_id,
                facts=tuple(members),
                resolution="surface_both" if len(members) > 1 and score_fact(ranked[0]) - score_fact(ranked[1]) < 0.08 else "prefer_higher_authority",
                winner_fact_id=None if len(members) > 1 and score_fact(ranked[0]) - score_fact(ranked[1]) < 0.08 else top.fact_id,
                reason="Scores too close; surface uncertainty" if len(members) > 1 and score_fact(ranked[0]) - score_fact(ranked[1]) < 0.08 else "Higher-ranked fact selected",
            )
        )

    return survivors, conflicts
```

### context_compressor.py

```python
from __future__ import annotations

from .context_budget import BudgetPolicy
from .context_ranker import rank_facts
from .synthesis_context import ContextFact, PromptContext
from .conflict_resolver import resolve_conflicts


def assemble_prompt_context(
    facts: list[ContextFact],
    budget: BudgetPolicy,
    unknowns: list[str] | None = None,
) -> PromptContext:
    ranked = rank_facts(facts)
    resolved, conflicts = resolve_conflicts(ranked)

    used_tokens = 0
    per_source_used: dict[str, int] = {}
    top_facts: list[ContextFact] = []
    supporting_facts: list[ContextFact] = []
    excluded: list[str] = []

    prompt_budget = budget.total_tokens - budget.reserve_for_answer

    for fact in resolved:
        cap = budget.per_source_caps.get(fact.source_class, prompt_budget)
        source_used = per_source_used.get(fact.source_class, 0)

        if source_used + fact.token_cost > cap:
            excluded.append(fact.fact_id)
            continue
        if used_tokens + fact.token_cost > prompt_budget:
            excluded.append(fact.fact_id)
            continue

        if len(top_facts) < budget.max_top_facts:
            top_facts.append(fact)
        elif len(supporting_facts) < budget.max_supporting_facts:
            supporting_facts.append(fact)
        else:
            excluded.append(fact.fact_id)
            continue

        used_tokens += fact.token_cost
        per_source_used[fact.source_class] = source_used + fact.token_cost

    return PromptContext(
        mode="synthesis",
        top_facts=tuple(top_facts),
        supporting_facts=tuple(supporting_facts),
        conflicts=tuple(conflicts),
        unknowns=tuple(unknowns or []),
        excluded_facts=tuple(excluded),
    )
```
