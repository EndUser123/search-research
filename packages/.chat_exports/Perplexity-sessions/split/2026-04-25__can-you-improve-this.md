#!/usr/bin/env python3
# Build structured gitpack output: TOC + signatures + full appendix

import re
import sys
from datetime import datetime, timezone
from pathlib import Path
from typing import Dict, List

def parse_aid_output(content: str) -> Dict[str, Dict[str, str]]:
    """Parse aid output into filepath -> {'sig': str, 'raw': str}."""
    file_pattern = re.compile(r'^(.+\.(py|ts|js|go|rs|java|cs|kt|cpp|c|h|php|hpp|rb|swift))(?:\s+python)?', re.MULTILINE)
    sections = {}
    matches = list(file_pattern.finditer(content))
    for i, m in enumerate(matches):
        filepath = m.group(1).strip()
        start = m.end(1) + 1  # Skip newline
        end = matches[i + 1].start() if i + 1 < len(matches) else len(content)
        section_text = content[start:end].strip()
        sections[filepath] = {'raw': section_text}
    return sections

def extract_signature(filepath: str, raw_section: str) -> str:
    """Extract signature: full for --quick, def/class/imports for full mode."""
    lines = raw_section.splitlines()
    stripped_lines = [line.strip() for line in lines]
    has_def_lines = any(
        stripped.startswith(kw) for line in stripped_lines for kw in ('def ', 'class ', 'async def ')
    )
    sig_lines = []
    in_code = False
    for line in lines:
        stripped = line.strip()
        if stripped.startswith('#') or stripped.startswith('"""') or stripped.startswith("'''"):
            in_code = True
        elif stripped.startswith('python') or stripped.startswith('```'):
            in_code = True
        elif not in_code:
            continue
        if has_def_lines:  # Full mode
            if any(stripped.startswith(kw) for kw in ('def ', 'class ', 'async def ')):
                sig_lines.append(line)
            elif stripped.startswith('import ') or stripped.startswith('from '):
                sig_lines.append(line)
            elif sig_lines and not stripped.endswith(':') and not stripped.startswith('#') and not stripped.startswith('@'):
                # Stop at impl body
                break
        else:  # --quick: collect func-like lines
            if stripped.startswith('import ') or stripped.startswith('from '):
                continue
            if ' -> ' in stripped or ' - ' in stripped:
                sig_lines.append(line)
    return '\n'.join(sig_lines).strip()

def group_by_dir(files: List[str]) -> Dict[str, List[str]]:
    groups = {}
    for filepath in files:
        parts = filepath.replace('\\', '/').split('/')
        top = parts if parts else filepath
        groups.setdefault(top, []).append(filepath)
    return groups

def build_structured_sig_sections(sig_sections: Dict[str, str], full_sections: Dict[str, str], dirname: str) -> str:
    all_files = list(full_sections.keys())
    groups = group_by_dir(all_files)
    total = len(all_files)

    # SIGNATURE TOC
    sig_lines = ['## SIGNATURE TOC']
    for filepath in sorted(all_files):
        sig_raw = sig_sections.get(filepath, {}).get('raw', '')
        sig = extract_signature(filepath, sig_raw)
        sig_lines.append(f'### {filepath}')
        sig_lines.append('```python')
        sig_lines.append(sig)
        sig_lines.append('```')
        sig_lines.append('')

    # DIRECTORY INDEX
    dir_lines = ['## DIRECTORY INDEX\n\n| Directory | Files |\n|-----------|-------|']
    for dir_name in sorted(groups.keys()):
        dir_lines.append(f'| {dir_name} | {len(groups[dir_name])} |')

    # FILE INDEX
    file_lines = ['## FILE INDEX\n\n| File | Description |\n|-------------------|--------------|']
    for filepath in sorted(all_files):
        name = Path(filepath).stem
        parts = filepath.replace('\\', '/').split('/')
        if name in ('__init__', 'index', 'main'):
            desc = f"Package {parts}" if len(parts) > 1 else filepath
        else:
            desc = name.replace('_', ' ').replace('-', ' ')
        file_lines.append(f"| {filepath} | {desc} |")

    # APPENDIX
    appendix_lines = ['---\n\n## APPENDIX FULL IMPLEMENTATIONS']
    for filepath in sorted(all_files):
        raw = full_sections.get(filepath, {}).get('raw', '')
        appendix_lines.append(f'### {filepath}')
        appendix_lines.append('```python')
        appendix_lines.append(raw)
        appendix_lines.append('```')
        appendix_lines.append('')

    # Header
    header = '\n'.join([
        f'# {dirname} LLM-READY PACK',
        '',
        '<!-- Generated by gitpackstructured.py (IMPROVED) -->',
        '',
        '## PACK INFO',
        f'- Files: {total} files',
        '- Mode: structured signatures + full appendix',
        f'- Generated: {datetime.now(timezone.utc).strftime("%Y-%m-%d %H:%M UTC")}',
        '',
        '## HOW TO USE THIS PACK',
        '1. SIGNATURE TOC: scan all file signatures to find relevant code',
        '2. FILE INDEX: jump to specific files by name',
        '3. APPENDIX FULL IMPLEMENTATIONS: read full implementation on demand',
        '',
        'For token efficiency: start with the SIGNATURE TOC, pull full code from the APPENDIX only when you need the implementation details.',
        ''
    ])

    return '\n'.join([header, '\n'.join(sig_lines), '\n'.join(dir_lines), '\n'.join(file_lines), '\n'.join(appendix_lines)])

def main() -> None:
    if len(sys.argv) != 4:
        print("Usage: gitpackstructured.py <sigoutput.md> <fulloutput.md> <dirname>", file=sys.stderr)
        sys.exit(1)
    sig_path = Path(sys.argv[1])
    full_path = Path(sys.argv)
    dirname = sys.argv
    if not sig_path.exists():
        print(f"ERROR: Signature file not found: {sig_path}", file=sys.stderr)
        sys.exit(1)
    if not full_path.exists():
        print(f"ERROR: Full output file not found: {full_path}", file=sys.stderr)
        sys.exit(1)

    sig_content = sig_path.read_text(encoding='utf-8')
    full_content = full_path.read_text(encoding='utf-8')
    sig_sections = parse_aid_output(sig_content)
    full_sections = parse_aid_output(full_content)
    sig_count = len(sig_sections)
    full_count = len(full_sections)
    if sig_count == 0:
        print("WARNING: No file signatures found in --quick output", file=sys.stderr)
    if full_count == 0:
        print("WARNING: No files found in full output", file=sys.stderr)
        sys.exit(1)

    # Mismatch check
    sig_files = set(sig_sections.keys())
    full_files = set(full_sections.keys())
    missing_in_full = sig_files - full_files
    extra_in_full = full_files - sig_files
    if missing_in_full:
        print(f"WARNING: {len(missing_in_full)} files in signatures but not in full output", file=sys.stderr)
    if extra_in_full:
        print(f"WARNING: {len(extra_in_full)} files in full output but not in signatures", file=sys.stderr)

    structured = build_structured_sig_sections(sig_sections, full_sections, dirname)
    full_path.write_text(structured, encoding='utf-8')
    print(f"Structured pack written: {full_count} files, {len(structured)} chars")
    if sig_count != full_count:
        print(f"Note: {sig_count} signatures, {full_count} full sections")

if __name__ == '__main__':
    main()
```

#!/usr/bin/env python3
# Prepend structured TOC/index to aid distillation output

import re
import sys
from datetime import datetime, timezone
from pathlib import Path
from typing import List, Tuple, Dict

def extract_files_content(content: str) -> List[Tuple[str, str]]:
    """Extract filepath/desc pairs from aid output."""
    pattern = r'^(.+\.(py|ts|js|go|rs|java|cs|kt|cpp|c|h|php|hpp|rb|swift))(?:\s+python)?'
    files: List[Tuple[str, str]] = []
    for line in content.splitlines():
        m = re.match(pattern, line.strip())
        if m:
            filepath = m.group(1).strip()
            files.append((filepath, describe_file(filepath)))
    return files

def describe_file(filepath: str) -> str:
    """Infer brief desc from filepath."""
    name = Path(filepath).stem
    parts = filepath.split('-')[1].split('-') if len(filepath.split('-')) > 2 else filepath.split('/')
    if name in ('__init__', 'index', 'main'):
        return f"Package {parts}" if len(parts) > 1 else filepath
    return name.replace('_', ' ').replace('-', ' ')

def group_by_dir_files(files: List[Tuple[str, str]]) -> Dict[str, List[str]]:
    groups = {}
    for filepath, _ in files:
        parts = filepath.replace('\\', '/').split('/')
        top = parts if parts else filepath
        groups.setdefault(top, []).append(filepath)
    return groups

def build_toc_files(files: List[Tuple[str, str]], dirname: str, mode: str) -> str:
    """Build TOC section."""
    groups = group_by_dir_files(files)
    total = len(files)
    lines = [f'# {dirname} LLM-READY PACK']
    lines += [
        '<!-- TOC is prepended by gitpacktoc.py -->',
        '',
        '## PACK INFO',
        f'- Files: {total} files distilled',
        f'- Mode: {mode}',
        f'- Generated: {datetime.now(timezone.utc).strftime("%Y-%m-%d %H:%M UTC")}',
        '',
        '## HOW TO USE THIS PACK',
        'This file is organized by directory. Use your LLM\'s search or jump-to-section to find relevant code.',
        'Each pathtofile.py header is a jump anchor.',
        '',
        'For token efficiency: read only sections relevant to your task.',
        '',
        '## DIRECTORY INDEX',
        '',
        '| Directory | Files |',
        '|-----------|-------|',
    ]
    for dir_name in sorted(groups.keys()):
        lines.append(f"| {dir_name} | {len(groups[dir_name])} |")
    lines += [
        '',
        '## FILE INDEX',
        '',
        '| File | Description |',
        '|-------------------|--------------|',
    ]
    for filepath, desc in sorted(files, key=lambda x: x):
        lines.append(f"| {filepath} | {desc} |")
    lines += ['', '---']
    return '\n'.join(lines)

def main() -> None:
    if len(sys.argv) < 3:
        print("Usage: gitpacktoc.py <aidoutput.md> <dirname> [mode]", file=sys.stderr)
        sys.exit(1)
    aid_output = Path(sys.argv[1])
    dirname = sys.argv
    mode = sys.argv if len(sys.argv) > 3 else 'full fidelity content'
    content = aid_output.read_text(encoding='utf-8')
    files = extract_files_content(content)
    if not files:
        print("WARNING: No files found in aid output. TOC may be empty.", file=sys.stderr)
        toc = build_toc('', dirname, mode)
    else:
        toc = build_toc_files(files, dirname, mode)
    new_content = toc + '\n' + content
    aid_output.write_text(new_content, encoding='utf-8')
    total = len(files)
    print(f"TOC prepended: {total} files indexed in {len(content)} chars of content")

if __name__ == '__main__':
    main()
```
