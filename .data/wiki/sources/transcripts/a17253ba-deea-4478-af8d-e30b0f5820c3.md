ADR-20260322: Hook Intent Detection Architecture

Status: Proposed

 

Date: 2026-03-22

 

Context: Cognitive enhancers hook fails to emit tags when prompts don't match rigid regex patterns, reducing framework

 

effectiveness.

Decision

Replace regex-based intent detection with a multi-layered detection architecture using:

Pattern-based layer (keep existing regex as fast path)

LLM-based classification layer (for ambiguous prompts)

Confidence scoring (only emit tags when detection is confident)

Graceful degradation (fallback to no tags rather than false positives)

Rationale

Current problem: Regex patterns are too rigid — legitimate implementation prompts that don't contain trigger words (e.g.,

 

"enhance the session resume mechanism") get no cognitive framework tags, reducing the value of the cognitive enhancers

 

system.

Why multi-layered works better:

Pattern layer provides zero-cost fast path for obvious cases

LLM layer handles nuance and context that regex cannot

Confidence scoring prevents bad tags from low-confidence detections

Graceful degradation maintains system integrity (no tags > wrong tags)

from enum import Enum

 

from dataclasses import dataclass

 

from typing import Optional, Pattern

 

import re

class IntentType(Enum):

 

IMPLEMENTATION = "implementation"

 

DIAGNOSTIC = "diagnostic"

 

DECOMPOSITION = "decomposition"

 

UNKNOWN = "unknown"

@dataclass

 

class IntentResult:

 

intent: IntentType

 

confidence: float  # 0.0 to 1.0

 

method: str  # "regex" | "llm" | "fallback"

class IntentClassifier:

 

"""Multi-layered intent detection for hooks."""

  def __init__(self, llm_client=None):
      self.llm_client = llm_client
      # Fast path patterns (existing regex)
      self.patterns = {
          IntentType.IMPLEMENTATION: re.compile(
              r'\b(build|create|implement|refactor|optimize|add|write|develop|code|make|set


up|configure|change|modify|update|fix|replace|rewrite|convert|migrate|hook up|wire up|integrate|extend|extract)\b',

 

re.IGNORECASE

 

),

 

IntentType.DIAGNOSTIC: re.compile(

 

r'\b(debug|investigate|diagnose|analyze|explain why|root cause|figure out|what's wrong|what

 

caused|troubleshoot|why does|why is|why did|how does|what happens)\b',

 

re.IGNORECASE

 

),

 

}

  def classify(self, prompt: str, threshold: float = 0.7) -> IntentResult:
      """Classify prompt intent using multi-layered detection."""

      # Layer 1: Pattern-based fast path (zero cost)
      pattern_result = self._classify_by_pattern(prompt)
      if pattern_result.confidence >= threshold:
          return pattern_result

      # Layer 2: LLM-based classification (for ambiguous prompts)
      if self.llm_client:
          llm_result = self._classify_by_llm(prompt)
          if llm_result.confidence >= threshold:
              return llm_result

      # Layer 3: Fallback (unknown intent)
      return IntentResult(IntentType.UNKNOWN, 0.0, "fallback")

  def _classify_by_pattern(self, prompt: str) -> IntentResult:
      """Fast path: regex-based classification."""
      for intent, pattern in self.patterns.items():
          if pattern.search(prompt):
              return IntentResult(intent, 0.9, "regex")  # High confidence for direct matches
      return IntentResult(IntentType.UNKNOWN, 0.0, "regex")

  def _classify_by_llm(self, prompt: str) -> IntentResult:
      """LLM-based classification for ambiguous prompts."""
      if not self.llm_client:
          return IntentResult(IntentType.UNKNOWN, 0.0, "llm_unavailable")

      classification_prompt = f"""Classify this prompt's intent:


Prompt: "{prompt}"

Respond with exactly one line:

 

INTENT:<intent_type> CONFIDENCE:<0.0-1.0>

Intent types:

implementation: Creating, modifying, or building code/systems

diagnostic: Investigating, debugging, or analyzing problems

decomposition: Breaking down complex problems

unknown: None of the above

Example responses:

 

INTENT:implementation CONFIDENCE:0.8

 

INTENT:diagnostic CONFIDENCE:0.6

 

INTENT:unknown CONFIDENCE:0.9

 

"""

      try:
          response = self.llm_client.generate(
              classification_prompt,
              max_tokens=50,
              temperature=0.0  # Deterministic
          )

          # Parse response
          intent_match = re.search(r'INTENT:(\w+)', response)
          conf_match = re.search(r'CONFIDENCE:([\d.]+)', response)

          if intent_match and conf_match:
              intent_str = intent_match.group(1).upper()
              confidence = float(conf_match.group(1))

              try:
                  intent = IntentType[intent_str]
                  return IntentResult(intent, confidence, "llm")
              except KeyError:
                  pass
      except Exception:
          pass

      return IntentResult(IntentType.UNKNOWN, 0.0, "llm_failed")

def userpromptsubmit_cognitive_enhancements(prompt: str, context: dict) -> str:

 

"""Emit cognitive framework tags based on detected intent."""

  # Load configuration
  config = load_cognitive_config()  # existing function
  if not config.get("enabled", False):
      return ""  # Disabled

  # Check suppressions (existing logic)
  if is_suppressed(context):
      return ""

  # Initialize classifier
  llm_client = get_llm_client() if config.get("use_llm_classification", True) else None
  classifier = IntentClassifier(llm_client)

  # Multi-layered classification
  result = classifier.classify(
      prompt,
      threshold=config.get("confidence_threshold", 0.7)
  )

  # Map intent to cognitive framework tags
  intent_to_tags = {
      IntentType.IMPLEMENTATION: ["[ASUM]", "[INV]", "[CAL]"],
      IntentType.DIAGNOSTIC: ["[ANCH]", "[FENC]", "[DADV]"],
      IntentType.DECOMPOSITION: ["[SOC]", "[CYNE]", "[RAZR]"],
  }

  if result.intent in intent_to_tags and result.confidence >= 0.7:
      tags = intent_to_tags[result.intent]
      return " " + " ".join(tags) + " "

  return ""  # No tags for unknown/low-confidence intents
