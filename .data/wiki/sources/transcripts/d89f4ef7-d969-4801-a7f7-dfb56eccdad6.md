import json

 

from pathlib import Path

 

import shutil

class TerminalStateManager:

 

"""Terminal-isolated state manager for Ralph loops."""

  def __init__(self, terminal_id: str, session_id: str):
      # Per-terminal state directory (your TASK-005 work!)
      self.state_dir = Path(
          f".claude/state/terminal_{terminal_id}/session_{session_id}"
      )
      self.state_file = self.state_dir / "loop_state.json"
      self.lock_file = self.state_dir / "loop_state.lock"

  def read_state(self) -> dict:
      """Atomically read state with lock checking."""
      if not self.state_file.exists():
          return {"tasks": [], "completed": [], "current": None}

      # Simple lock check (no waiting - if locked, return stale)
      if self.lock_file.exists():
          return {"tasks": [], "completed": [], "current": None, "stale": True}

      try:
          with open(self.state_file) as f:
              return json.load(f)
      except (json.JSONDecodeError, IOError):
          return {"tasks": [], "completed": [], "current": None, "error": True}

  def write_state(self, state: dict) -> bool:
      """Atomically write state with file locking."""
      self.state_dir.mkdir(parents=True, exist_ok=True)

      # Write to temp file, then atomic rename
      temp_file = self.state_file.with_suffix('.tmp')
      try:
          with open(temp_file, 'w') as f:
              json.dump(state, f, indent=2)

          # Atomic rename (Windows requires overwrite)
          temp_file.replace(self.state_file)
          return True
      except (IOError, OSError):
          return False


Why this works for multi-terminal:

Each terminal has its own state directory (TASK-005 work)

No shared Git state to race on

Atomic writes via temp file + rename

Lock file prevents concurrent writes (same terminal only)

If lock exists, return "stale" flag (caller decides whether to wait)

def parse_plan_tasks(plan_path: Path) -> list[dict]:

 

"""Parse plan.md for task completion state."""

  tasks = []
  for line in content.split('\n'):
      if line.strip().startswith('- ['):
          completed = line.strip().startswith('- [x]')
          task_text = line.strip().split(']', 1)[1].strip()

  return tasks
