import argparse
import ast
import json
import os
import re
import sys

class ASTSecurityVisitor(ast.NodeVisitor):
    """AST visitor to detect insecure code execution patterns within Python tools."""
    def __init__(self, file_path):
        self.file_path = file_path
        self.findings =

    def visit_Call(self, node):
        # Identify unsafe subprocess invocations with shell=True
        if isinstance(node.func, ast.Attribute) and node.func.attr in ["call", "Popen", "run"]:
            for keyword in node.keywords:
                if keyword.arg == "shell" and isinstance(keyword.value, ast.Constant) and keyword.value.value is True:
                    self.findings.append({
                        "file": self.file_path,
                        "line": node.lineno,
                        "severity": "CRITICAL",
                        "type": "Command Injection",
                        "details": f"Subprocess invocation with shell=True detected in {ast.dump(node.func)}"
                    })
        # Identify native code execution calls
        elif isinstance(node.func, ast.Name) and node.func.id in ["eval", "exec", "compile"]:
            self.findings.append({
                "file": self.file_path,
                "line": node.lineno,
                "severity": "CRITICAL",
                "type": "Arbitrary Code Execution",
                "details": f"Dangerous usage of built-in function '{node.func.id}()'"
            })
        self.generic_visit(node)

def run_signature_scan(file_path):
    """Performs raw signature and regex audits on scripts."""
    findings =
    try:
        with open(file_path, "r", encoding="utf-8") as f:
            content = f.read()
            
        # Match signatures against defined vulnerability profiles
        for category, patterns in CRITICAL_SIGNATURES.items():
            for pattern in patterns:
                for match in re.finditer(pattern, content):
                    line_no = content.count("\n", 0, match.start()) + 1
                    findings.append({
                        "file": file_path,
                        "line": line_no,
                        "severity": "CRITICAL",
                        "type": category.upper(),
                        "details": f"Matched malicious signature pattern: '{pattern}'"
                    })
    except Exception as e:
        findings.append({
            "file": file_path,
            "line": 0,
            "severity": "MEDIUM",
            "type": "File Read Exception",
            "details": f"Unable to read file contents: {str(e)}"
        })
    return findings

def audit_skill_directory(skill_path):
    """Walks the target directory and scans all matching files."""
    all_findings =
    for root, _, files in os.walk(skill_path):
        for file in files:
            file_path = os.path.join(root, file)
            if file.endswith(".py"):
                # Run both structural and regex scans on Python scripts
                try:
                    with open(file_path, "r", encoding="utf-8") as f:
                        tree = ast.parse(f.read(), filename=file_path)
                    visitor = ASTSecurityVisitor(file_path)
                    visitor.visit(tree)
                    all_findings.extend(visitor.findings)
                except Exception as e:
                    all_findings.append({
                        "file": file_path,
                        "line": 0,
                        "severity": "HIGH",
                        "type": "AST Parse Failure",
                        "details": f"Failed to construct AST tree: {str(e)}"
                    })
                all_findings.extend(run_signature_scan(file_path))
            elif file.endswith((".sh", ".js")):
                all_findings.extend(run_signature_scan(file_path))
    return all_findings

def main():
    parser = argparse.ArgumentParser(description="Static security scanner for custom agent skills")
    parser.add_argument("skill_dir", help="Target skill directory path to analyze")
    parser.add_argument("--strict", action="store_true", help="Fail on any warnings")
    parser.add_argument("--json", action="store_true", help="Output findings as JSON")
    args = parser.parse_args()

    findings = audit_skill_directory(args.skill_dir)
    has_critical = any(f["severity"] == "CRITICAL" for f in findings)
    
    if args.json:
        print(json.dumps({
            "verdict": "FAIL" if has_critical else "PASS",
            "findings": findings
        }, indent=2))
    else:
        for f in findings:
            print(f"[{f['severity']}] {f['file']}:{f['line']} - {f['type']} - {f['details']}")
            
    if has_critical:
        sys.exit(2)
    elif len(findings) > 0 and args.strict:
        sys.exit(1)
    sys.exit(0)

if __name__ == "__main__":
    main()

    gaps = data.get("gaps",)
    if not isinstance(gaps, list) or len(gaps) < 3:
        return {"status": "FAIL", "reason": "Insufficient analysis depth: GTO report must identify at least three structural gaps."}

    debt_index = float(data.get("debt_index", 0.0))
    remediation_time = float(data.get("remediation_time_hours", 0.0))

    if debt_index <= 0.0:
        return {"status": "FAIL", "reason": "Technical debt index must be a positive float value."}
    if remediation_time <= 0.0:
        return {"status": "FAIL", "reason": "Estimated remediation timeline must be greater than zero."}

    cumulative_opportunities = 0.0
    for idx, gap in enumerate(gaps):
        # Extract evaluation attributes for each gap
        impact = float(gap.get("impact_factor", 1.0)) # G_j
        feasibility = float(gap.get("feasibility_coefficient", 0.5)) # C_j
        
        if not (1.0 <= impact <= 10.0):
            return {"status": "FAIL", "reason": f"Impact factor for gap {idx} must be in range . Got: {impact}"}
        if not (0.0 < feasibility <= 1.0):
            return {"status": "FAIL", "reason": f"Feasibility coefficient for gap {idx} must be in range (0, 1.0]. Got: {feasibility}"}
            
        cumulative_opportunities += (impact * feasibility)

    # Calculate Opportunity Value Index (OVI)
    ovi = cumulative_opportunities / (remediation_time * math.log(1.0 + debt_index))

    if ovi < LAMBDA_THRESHOLD:
        return {
            "status": "FAIL",
            "reason": f"Calculated Opportunity Value Index (OVI: {ovi:.3f}) falls below the required threshold (lambda: {LAMBDA_THRESHOLD}). Refine remediation efficiency."
        }

    return {
        "status": "PASS",
        "ovi": ovi,
        "gaps_scanned": len(gaps)
    }

def main():
    parser = argparse.ArgumentParser(description="Deterministic Quality Gate for GTO Analysis")
    parser.add_argument("--report", default="GTO_REPORT.json", help="Path to the target GTO report JSON file")
    parser.add_argument("--json", action="store_true", help="Output validation results as structured JSON")
    args = parser.parse_args()

    if not os.path.exists(args.report):
        results = {"status": "FAIL", "reason": "GTO report file GTO_REPORT.json is missing."}
    else:
        try:
            with open(args.report, "r", encoding="utf-8") as f:
                data = json.load(f)
            results = calculate_gto_metrics(data)
        except json.JSONDecodeError as e:
            results = {"status": "FAIL", "reason": f"GTO_REPORT.json contains invalid JSON: {str(e)}"}
        except Exception as e:
            results = {"status": "FAIL", "reason": f"An unexpected system exception occurred: {str(e)}"}

    if results["status"] == "FAIL":
        if args.json:
            print(json.dumps(results, indent=2))
        else:
            sys.stderr.write(f"[-] GTO QUALITY GATE ENFORCEMENT FAILURE:\nReason: {results['reason']}\n")
        # Exit code 2 triggers active enforcement on blockable events
        sys.exit(2)

    if args.json:
        print(json.dumps(results, indent=2))
    else:
        print(f"[+] GTO QUALITY GATE PASSED: Opportunity Value Index is {results['ovi']:.3f}")
    sys.exit(0)

if __name__ == "__main__":
    main()

: The

 hook runs before any filesystem-modifying tool executes.[7, 18] Finding no valid

Forced GTO Execution

: The agent processes the block, invokes the GTO skill, runs the local analysis tool, and generates a structured 

GTO_REPORT.json

 file.[12, 13]

Turn Verification

: When the agent attempts to conclude its turn, the 

Stop

 event triggers the quality gate script.[16, 18] The script parses the report and evaluates the GTO metrics against the target threshold.[16]

Self-Correction Loop

: If the report contains incomplete data or fails to meet the required 

OVI

 threshold, the evaluator exits with code 2, sending the validation errors back to the model as an execution failure.[10, 15] The agent reads these errors as its new context, recalculates the opportunity ratios, updates the 

GTO_REPORT.json

 file, and re-submits the analysis.[2, 7, 15] This loop continues until all programmatic checks pass, guaranteeing that every codebase modification is backed by a valid, high-rigor GTO report.[7, 15]
