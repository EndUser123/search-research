async def _compute_idea_confidence(self, idea: Idea) -> float:

 

"""Compute agent's confidence in generated idea (0-1 scale).

  Factors:
  - Idea specificity (length, detail level)
  - Internal consistency (no contradictions)
  - Relevance to context (addressed core constraints)
  - Uniqueness (not duplicating prior ideas)
  """
  prompt = f"""
  Rate your confidence in this idea (0.0-1.0):

  Idea: {idea.title}
  Description: {idea.description}

  Consider:
  - Specificity: Is the idea detailed and actionable?
  - Consistency: Are there internal contradictions?
  - Relevance: Does it address the core problem?
  - Uniqueness: Is this distinct from prior ideas?

  Return only a numeric score (0.0-1.0).
  """
  response = await self.llm_client.generate(
      prompt=prompt,
      temperature=0.3,  # Low temperature for consistent scoring
      max_tokens=10,
  )
  try:
      score = float(response.content.strip())
      return max(0.0, min(1.0, score))  # Clamp to [0,1]
  except (ValueError, IndexError):
      return 0.5  # Default on parsing failure


Step 3: Add confidence-weighted scheduling to lib/orchestrator.py

Add to BrainstormOrchestrator class

async def _schedule_confidence_based_turns(

 

self,

 

ideas_awaiting_turn: list[tuple[Agent, Idea]],

 

phase: str  # "diverge" | "discuss"

 

) -> list[tuple[Agent, Idea]]:

 

"""Schedule agent turns based on confidence thresholds.

  Diverge phase: High-confidence agents (≥0.7) go first to establish foundations
  Discuss phase: Mid-confidence (0.4-0.7) dominate for balanced evaluation
  Wildcard slot: Low-confidence (<0.4) saved for late-phase novelty injection
  """
  if phase == "diverge":
      # Sort by confidence descending (high first)
      return sorted(ideas_awaiting_turn, key=lambda x: x[1].confidence, reverse=True)
  elif phase == "discuss":
      # Mid-confidence first, then high, then low (wildcard)
      mid_conf = [x for x in ideas_awaiting_turn if 0.4 <= x[1].confidence < 0.7]
      high_conf = [x for x in ideas_awaiting_turn if x[1].confidence >= 0.7]
      low_conf = [x for x in ideas_awaiting_turn if x[1].confidence < 0.4]
      return mid_conf + high_conf + low_conf
  else:
      # Converge phase: default round-robin
      return ideas_awaiting_turn
