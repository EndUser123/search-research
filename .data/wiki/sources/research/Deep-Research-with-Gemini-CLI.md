```
function Invoke-VideoResearch {
    param(
        [Parameter(Mandatory=$true, Position=0)]
```

```
function Remove-OldResearch {
    param(
        [int]$Days = 7,
        [switch]$Force
    )

    $targetFiles = Get-ChildItem "research_*.md" | Where-Object { 
        $_.LastWriteTime -lt (Get-Date).AddDays(-$Days) 
    }

    if ($targetFiles.Count -eq 0) {
        Write-Host "No research reports older than $Days days found." -ForegroundColor Gray
        return
    }

    if ($Force) {
        $targetFiles | Remove-Item -Verbose
        Write-Host "Cleaned up $($targetFiles.Count) old research reports." -ForegroundColor Green
    } else {
        Write-Host "The following $($targetFiles.Count) files would be deleted (use -Force to confirm):" -ForegroundColor Yellow
        $targetFiles | Select-Object Name, LastWriteTime | Format-Table
    }
}
```

## 1\. The `cks.py` Framework

This script serves as the backbone for Category 12. It takes the raw output from your `gemini-cli` video analysis and "promotes" it into your agent's long-term memory.

Python

```
import json
import re
from pathlib import Path
from typing import List, Optional, Dict
from datetime import datetime
from pydantic import BaseModel, Field
from rich.console import Console
from rich.table import Table

console = Console()

# The 12 categories of the Cognitive Steering Framework
CSF_CATEGORIES = {
    1: "Truth & Evidence",
    2: "Quality & Behavioral",
    3: "Constitutional Patterns",
    4: "Code Protection",
    5: "Investigation & Research",
    6: "Path & File Safety",
    7: "Task & Session Management",
    8: "Cognitive Frameworks",
    9: "Skill & Workflow Enforcement",
    10: "Observability & Logging",
    11: "Formatting & Style",
    12: "CKS (Constitutional Knowledge System)"
}

class CSFArtifact(BaseModel):
    category_id: int
    title: str
    content: str
    source_url: Optional[str] = None
    timestamp: datetime = Field(default_factory=datetime.now)
    tags: List[str] = []

class ConstitutionalKnowledgeSystem:
    def __init__(self, storage_path: str = "./.cks/registry.json"):
        self.storage_path = Path(storage_path)
        self.storage_path.parent.mkdir(exist_ok=True)
        self.artifacts: List[CSFArtifact] = self._load_data()

    def _load_data(self) -> List[CSFArtifact]:
        if self.storage_path.exists():
            with open(self.storage_path, "r") as f:
                data = json.load(f)
                return [CSFArtifact(**item) for item in data]
        return []

    def save(self):
        with open(self.storage_path, "w") as f:
            json.dump([a.model_dump(mode='json') for a in self.artifacts], f, indent=2)

    def ingest_report(self, file_path: Path):
        """Parses a gemini-cli research markdown file for CSF data."""
        content = file_path.read_text()
        
        # Regex to find blocks like: [CSF-3] Reflexion Pattern
        # This allows the gemini-cli to 'tag' things in the video
        matches = re.finditer(r"\[CSF-(\d+)\]\s*(.*?)\n(.*?)(?=\[CSF-|\Z)", content, re.DOTALL)
        
        count = 0
        for match in matches:
            cat_id = int(match.group(1))
            title = match.group(2).strip()
            body = match.group(3).strip()
            
            artifact = CSFArtifact(
                category_id=cat_id,
                title=title,
                content=body,
                tags=["video-extraction"]
            )
            self.artifacts.append(artifact)
            count += 1
        
        self.save()
        return count

    def get_steering_prompt(self, category_id: int) -> str:
        """Returns all constitutional rules for a specific category."""
        relevant = [a.content for a in self.artifacts if a.category_id == category_id]
        return "\n---\n".join(relevant)

    def display_status(self):
        table = Table(title="Cognitive Steering Framework - CKS Status")
        table.add_column("ID", style="cyan")
        table.add_column("Category Name", style="magenta")
        table.add_column("Artifact Count", justify="right", style="green")

        for cat_id, name in CSF_CATEGORIES.items():
            count = len([a for a in self.artifacts if a.category_id == cat_id])
            table.add_row(str(cat_id), name, str(count))
        
        console.print(table)

# Example Usage
if __name__ == "__main__":
    cks = ConstitutionalKnowledgeSystem()
    # cks.ingest_report(Path("research_20260326.md"))
    cks.display_status()
```

```
# csf-watcher.ps1
$path = Get-Location
$filter = 'research_*.md'

$watcher = New-Object System.IO.FileSystemWatcher
$watcher.Path = $path
$watcher.Filter = $filter
$watcher.IncludeSubdirectories = $false
$watcher.EnableRaisingEvents = $true

$action = {
    $path = $Event.SourceEventArgs.FullPath
    $name = $Event.SourceEventArgs.Name
    
    Write-Host "`n[CSF-HOOK] New Research Detected: $name" -ForegroundColor Cyan
    Write-Host "[CSF-HOOK] Waiting for file lock to release..." -ForegroundColor Gray
    
    # Wait for Gemini to finish writing and release the file handle
    while ($true) {
        try {
            $file = [System.IO.File]::Open($path, 'Open', 'Read', 'None')
            $file.Close()
            break
        } catch {
            Start-Sleep -Milliseconds 500
        }
    }

    Write-Host "[CSF-HOOK] Ingesting into CKS (Category 12)..." -ForegroundColor Yellow
    python cks.py --ingest $path
    
    # Optional: Windows Notification
    # Add-Type -AssemblyName System.Windows.Forms
    # [System.Windows.Forms.MessageBox]::Show("CKS Updated: $name")
}

Register-ObjectEvent $watcher "Created" -Action $action
Write-Host "CSF Watcher is active. Monitoring for Deep Research artifacts..." -ForegroundColor Green
```

### 2\. The Core Framework: `cks.py`

Python

```
import json, re
from pathlib import Path
from typing import List, Dict
from datetime import datetime
from pydantic import BaseModel, Field
from rich.console import Console
from rich.table import Table

console = Console()

class CSFArtifact(BaseModel):
    category_id: int
    title: str
    content: str
    timestamp: datetime = Field(default_factory=datetime.now)

class ConstitutionalKnowledgeSystem:
    def __init__(self, storage="./.cks/registry.json"):
        self.storage = Path(storage)
        self.artifacts = self._load()

    def _load(self):
        if self.storage.exists():
            return [CSFArtifact(**a) for a in json.load(self.storage.open())]
        return []

    def ingest_report(self, path: Path):
        content = path.read_text()
        matches = re.finditer(r"\[CSF-(\d+)\]\s*(.*?)\n(.*?)(?=\[CSF-|\Z)", content, re.DOTALL)
        new_count = 0
        for m in matches:
            self.artifacts.append(CSFArtifact(category_id=int(m.group(1)), title=m.group(2).strip(), content=m.group(3).strip()))
            new_count += 1
        self.storage.write_text(json.dumps([a.model_dump(mode='json') for a in self.artifacts], indent=2))
        return new_count

    def get_diff(self, old_snapshot: List[CSFArtifact]):
        old_titles = {a.title for a in old_snapshot}
        return [a for a in self.artifacts if a.title not in old_titles]
```

```
def get_private_urls(source=":ytwatchlater"):

    """
    cmd = [

    ]
```

```
import asyncio
from pathlib import Path
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler
from cks import ConstitutionalKnowledgeSystem

class WatcherHandler(FileSystemEventHandler):
    def __init__(self):
        self.cks = ConstitutionalKnowledgeSystem()

    def on_closed(self, event):
        if not event.src_path.endswith('.md'): return
        file_path = Path(event.src_path)
        if file_path.name.startswith("research_"):
            snapshot = list(self.cks.artifacts)
            new = self.cks.ingest_report(file_path)
            if new > 0:
                print(f"\n[CKS UPGRADE] {new} new patterns found.")
                for diff in self.cks.get_diff(snapshot):
                    print(f"  - ({diff.category_id}) {diff.title}")

async def main():
    observer = Observer()
    observer.schedule(WatcherHandler(), ".", recursive=False)
    observer.start()
    print("Intelligence Stream Watcher Active...")
    try:
        while True: await asyncio.sleep(1)
    except: observer.stop()
    observer.join()

if __name__ == "__main__":
    asyncio.run(main())
```

```
class CKSRecovery:
    def __init__(self, cks_dir="./.cks"):

    def create_checkpoint(self, label: str = "auto"):
        """Creates a time-sortable backup using Python 3.14 UUIDv7."""
        # UUIDv7 is time-ordered, making 'latest' identification trivial
        checkpoint_id = uuid.uuid7() 
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        backup_path = self.backup_dir / t"checkpoint_{timestamp}_{checkpoint_id}.json"
        
        if self.registry.exists():
            # Python 3.14 pathlib now supports native .copy()
            self.registry.copy(backup_path)

            return backup_path
        return None

    def rollback(self):
        """Restores the most recent valid checkpoint."""
        # Get all checkpoints sorted by filename (since they use UUIDv7/Timestamps)

        # Atomic restore
        latest.copy(self.registry)

        return True
```

```
        registry = json.loads(self.registry_path.read_text())
        
        # 2. Analyze Growth (Last 7 Days)
        seven_days_ago = datetime.now() - timedelta(days=7)
        new_artifacts = [
            a for a in registry 
            if datetime.fromisoformat(a['timestamp']) > seven_days_ago
        ]
        
        # 3. Analyze Failures/Rollbacks
        failures = self.audit_log.read_text().splitlines() if self.audit_log.exists() else []
        recent_failures = [f for f in failures if "2026-03" in f] # Simple date filter

        # 4. Domain Distribution
        domain_counts = Counter([a['category_id'] for a in new_artifacts])
        
        # 5. Build the Briefing (Python 3.14 t-string)
        report_tmpl = Template(t"""
        # CSF WEEKLY STEERING BRIEFING
        Period: $start to $end
        
        ## 🧠 CONSTITUTIONAL GROWTH
        - New Patterns Ingested: $count
        - Primary Domain Focus: Domain $top_domain
        
        ## 🛡️ INTEGRITY & SAFETY
        - Successful Audits (Cat 7): $success_count
        - Blocked Injections/Rollbacks: $fail_count
        
        ## 🚀 RECOMMENDED STEERING
        Based on Category 8 (Drift), prioritize Domain 5 (Investigation) this week.
        """)
        
        report = report_tmpl.substitute(
            start=seven_days_ago.strftime("%Y-%m-%d"),
            end=datetime.now().strftime("%Y-%m-%d"),
            count=len(new_artifacts),
            top_domain=domain_counts.most_common(1)[0][0] if domain_counts else "N/A",
            success_count=len(new_artifacts),
            fail_count=len(recent_failures)
        )
        
        console.print(Panel(report, title="[bold cyan]Monday Morning Briefing[/bold cyan]", expand=False))

if __name__ == "__main__":
    briefing = MondayBriefing()
    briefing.generate_report()
```
