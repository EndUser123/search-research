## Measurement Tools and Automation

### 1. Automated Metrics Collection

#### Metrics Collection Framework
Create `.speckit/scripts/metrics-collector.py`:
```python
#!/usr/bin/env python3
"""
Automated metrics collection for Speckit
"""

import json
import time
import statistics
from datetime import datetime, timedelta
from pathlib import Path
from typing import Dict, List, Any

class SpeckitMetricsCollector:
    def __init__(self, config_file: str = ".speckit/config/metrics.json"):
        self.config = self.load_config(config_file)
        self.metrics_history = []

    def collect_all_metrics(self) -> Dict[str, Any]:
        """Collect comprehensive metrics"""
        timestamp = datetime.now()

        metrics = {
            "timestamp": timestamp.isoformat(),
            "productivity_metrics": self.collect_productivity_metrics(),
            "quality_metrics": self.collect_quality_metrics(),
            "collaboration_metrics": self.collect_collaboration_metrics(),
            "performance_metrics": self.collect_performance_metrics(),
            "adoption_metrics": self.collect_adoption_metrics()
        }

        self.metrics_history.append(metrics)
        self.save_metrics_history()

        return metrics

    def collect_productivity_metrics(self) -> Dict[str, Any]:
        """Collect productivity-related metrics"""
        # Analyze task completion rates
        tasks_file = Path(".speckit/cache/active_tasks.json")
        completed_tasks_file = Path(".speckit/cache/completed_tasks.json")

        metrics = {
            "task_completion_rate": 0,
            "average_task_duration": 0,
            "tasks_completed_per_week": 0,
            "onboarding_time": 0
        }

        if tasks_file.exists() and completed_tasks_file.exists():
            with open(tasks_file) as f:
                active_tasks = json.load(f)
            with open(completed_tasks_file) as f:
                completed_tasks = json.load(f)

            total_tasks = len(active_tasks) + len(completed_tasks)
            if total_tasks > 0:
                metrics["task_completion_rate"] = len(completed_tasks) / total_tasks

            # Calculate average task duration
            if completed_tasks:
                durations = []
                for task in completed_tasks:
                    if "created_date" in task and "completed_date" in task:
                        created = datetime.fromisoformat(task["created_date"])
                        completed = datetime.fromisoformat(task["completed_date"])
                        duration = (completed - created).total_seconds() / 3600  # hours
                        durations.append(duration)

                if durations:
                    metrics["average_task_duration"] = statistics.mean(durations)

        return metrics

    def collect_quality_metrics(self) -> Dict[str, Any]:
        """Collect quality-related metrics"""
        metrics = {
            "average_trust_score": 0,
            "test_coverage_percentage": 0,
            "defect_density": 0,
            "security_score": 0
        }

        # Analyze evidence files for quality metrics
        evidence_dir = Path(".speckit/evidence")
        if evidence_dir.exists():
            trust_scores = []
            for evidence_file in evidence_dir.rglob("*.json"):
                try:
                    with open(evidence_file) as f:
                        evidence = json.load(f)
                        if "trust_score" in evidence:
                            trust_scores.append(evidence["trust_score"])
                except:
                    continue

            if trust_scores:
                metrics["average_trust_score"] = statistics.mean(trust_scores)

        return metrics

    def collect_performance_metrics(self) -> Dict[str, Any]:
        """Collect performance metrics"""
        import psutil

        metrics = {
            "response_time_ms": 0,
            "memory_usage_mb": psutil.virtual_memory().used / (1024 * 1024),
            "cpu_usage_percent": psutil.cpu_percent(),
            "disk_usage_percentage": psutil.disk_usage('.').percent
        }

        # Measure Speckit response time
        start_time = time.time()
        try:
            result = subprocess.run(['speckit', '--version'],
                                  capture_output=True, text=True, timeout=10)
            if result.returncode == 0:
                end_time = time.time()
                metrics["response_time_ms"] = (end_time - start_time) * 1000
        except:
            metrics["response_time_ms"] = 0

        return metrics

    def generate_metrics_report(self, period_days: int = 30) -> Dict[str, Any]:
        """Generate comprehensive metrics report"""
        cutoff_date = datetime.now() - timedelta(days=period_days)

        # Filter metrics for the period
        period_metrics = [
            m for m in self.metrics_history
            if datetime.fromisoformat(m["timestamp"]) > cutoff_date
        ]

        if not period_metrics:
            return {"error": "No metrics available for the specified period"}

        report = {
            "report_period": f"{period_days} days",
            "report_generated": datetime.now().isoformat(),
            "summary": self.calculate_summary_statistics(period_metrics),
            "trends": self.calculate_trends(period_metrics),
            "recommendations": self.generate_recommendations(period_metrics)
        }

        return report

    def save_metrics_history(self):
        """Save metrics history to file"""
        metrics_file = Path(".speckit/evidence/metrics/metrics_history.json")
        metrics_file.parent.mkdir(parents=True, exist_ok=True)

        with open(metrics_file, 'w') as f:
            json.dump(self.metrics_history, f, indent=2, default=str)

    def calculate_summary_statistics(self, metrics: List[Dict[str, Any]]) -> Dict[str, Any]:
        """Calculate summary statistics from metrics data"""
        summary = {}

        # Productivity metrics summary
        completion_rates = [m["productivity_metrics"]["task_completion_rate"] for m in metrics if m["productivity_metrics"]["task_completion_rate"] > 0]
        if completion_rates:
            summary["avg_completion_rate"] = statistics.mean(completion_rates)
            summary["completion_rate_trend"] = "improving" if completion_rates[-1] > completion_rates[0] else "declining"

        # Quality metrics summary
        trust_scores = [m["quality_metrics"]["average_trust_score"] for m in metrics if m["quality_metrics"]["average_trust_score"] > 0]
        if trust_scores:
            summary["avg_trust_score"] = statistics.mean(trust_scores)
            summary["trust_score_trend"] = "improving" if trust_scores[-1] > trust_scores[0] else "declining"

        # Performance metrics summary
        response_times = [m["performance_metrics"]["response_time_ms"] for m in metrics if m["performance_metrics"]["response_time_ms"] > 0]
        if response_times:
            summary["avg_response_time_ms"] = statistics.mean(response_times)
            summary["response_time_trend"] = "improving" if response_times[-1] < response_times[0] else "declining"

        return summary

    def calculate_trends(self, metrics: List[Dict[str, Any]]) -> Dict[str, str]:
        """Calculate trend directions"""
        trends = {}

        if len(metrics) >= 2:
            # Compare first half with second half
            mid_point = len(metrics) // 2
            first_half = metrics[:mid_point]
            second_half = metrics[mid_point:]

            # Productivity trends
            first_completion_rate = statistics.mean([m["productivity_metrics"]["task_completion_rate"] for m in first_half if m["productivity_metrics"]["task_completion_rate"] > 0])
            second_completion_rate = statistics.mean([m["productivity_metrics"]["task_completion_rate"] for m in second_half if m["productivity_metrics"]["task_completion_rate"] > 0])

            if second_completion_rate > first_completion_rate * 1.05:
                trends["productivity"] = "improving"
            elif second_completion_rate < first_completion_rate * 0.95:
                trends["productivity"] = "declining"
            else:
                trends["productivity"] = "stable"

        return trends

    def generate_recommendations(self, metrics: List[Dict[str, Any]]) -> List[str]:
        """Generate improvement recommendations"""
        recommendations = []

        # Analyze latest metrics
        if metrics:
            latest = metrics[-1]

            if latest["productivity_metrics"]["task_completion_rate"] < 0.8:
                recommendations.append("Consider task breakdown improvements and better estimation practices")

            if latest["quality_metrics"]["average_trust_score"] < 0.8:
                recommendations.append("Focus on quality improvement initiatives and additional validation")

            if latest["performance_metrics"]["response_time_ms"] > 2000:
                recommendations.append("Optimize Speckit performance and consider cache optimization")

        if not recommendations:
            recommendations.append("All metrics are within acceptable ranges - continue current practices")

        return recommendations
```
